# Character references

HTML's character references, which both parsers need: named ones, such
as `&amp;` and `&ouml;`, from the table in `entities.temper.md`
(generated by `scripts/character_references.py`), and numeric ones,
such as `&#35;` and `&#x22;`.

Parsers keep a reference as written, in its own token, and decode it
only when asked, so the tree stays lossless.

```temper inert
namedCharacterReference("ouml;");    // "ö"
namedCharacterReference("amp");      // "&": HTML still takes a few without `;`
namedCharacterReference("ouml");     // null: not this one
decodeCharacterReference("&#x22;");  // "\""
```

## Named references

The table, read into a map when the library loads: each name with its
semicolon, and, for the 106 that HTML also accepts without one, the
name without it too.

    let namedReferences: Map<String, String> = readNamedReferences();

    let readNamedReferences(): Map<String, String> {
      let map = new MapBuilder<String, String>();
      for (var k = 0; k < namedReferenceChunks.length; k += 1) {
        let chunk = namedReferenceChunks[k];
        var previous = "";
        var i = String.begin;
        while (chunk.hasIndex(i)) {
          let shared = chunk[i] - 0x40;
          i = chunk.next(i);
          let suffixStart = i;
          while (isAsciiAlphanumeric(chunk[i])) { i = chunk.next(i); }
          let name = "${prefixOf(previous, shared)}${chunk.slice(suffixStart, i)}";
          let marker = chunk[i];
          i = chunk.next(i);
          let valueStart = i;
          i = chunk.next(i);
          if (marker == char':') { i = chunk.next(i); }
          let value = chunk.slice(valueStart, i);
          map["${name};"] = value;
          if (marker == char',') { map[name] = value; }
          previous = name;
        }
      }
      map.toMap()
    }

The first `count` code points of `text`, which is ASCII here.

    let prefixOf(text: String, count: Int): String {
      var i = String.begin;
      for (var n = 0; n < count; n += 1) { i = text.next(i); }
      text.slice(String.begin, i)
    }

What the named reference `name` stands for, or null. `name` is what
follows the `&`, with its semicolon if it has one: `"amp;"` and
`"amp"` both give `"&"`, since HTML accepts that one without its
semicolon, but `"ouml"` gives null. CommonMark takes only the forms
with a semicolon.

    export let namedCharacterReference(name: String): String? {
      let value = namedReferences.getOr(name, "");
      if (value.isEmpty) { null } else { value }
    }

## Numeric references

The code point `value` as text, or U+FFFD for one that isn't a Unicode
scalar value (0, a surrogate, or past U+10FFFF), as CommonMark decodes
numeric references. (HTML also remaps 0x80 to 0x9F, as Windows-1252;
kelpto-html will do that itself.)

    export let codePointOrReplacement(value: Int): String {
      let c = if (value <= 0 || value > 0x10FFFF || (value >= 0xD800 && value <= 0xDFFF)) {
        0xFFFD
      } else {
        value
      };
      String.fromCodePoint(c) orelse panic()
    }

## Finding and decoding

The end of the character reference starting with the `&` at `at`, by
CommonMark's rules, or -1: `&`, then a name in HTML's table, `#` and 1
to 7 decimal digits, or `#x` or `#X` and 1 to 6 hex digits, then `;`.

    export let characterReferenceEnd(source: SourceText, at: Int, end: Int): Int {
      if (at >= end || source.codeAt(at) != char'&') { return -1; }
      var i = at + 1;
      if (i < end && source.codeAt(i) == char'#') {
        i += 1;
        let hex = i < end && (source.codeAt(i) == char'x' || source.codeAt(i) == char'X');
        if (hex) { i += 1; }
        let digitsStart = i;
        while (
          i < end && i - digitsStart < 8 &&
          (if (hex) { isAsciiHexDigit(source.codeAt(i)) } else { isAsciiDigit(source.codeAt(i)) })
        ) {
          i += 1;
        }
        let digits = i - digitsStart;
        let most = if (hex) { 6 } else { 7 };
        if (digits < 1 || digits > most || i >= end || source.codeAt(i) != char';') { return -1; }
        return i + 1;
      }
      let nameStart = i;
      while (i < end && i - nameStart <= 32 && isAsciiAlphanumeric(source.codeAt(i))) { i += 1; }
      if (i == nameStart || i >= end || source.codeAt(i) != char';') { return -1; }
      if (!isAsciiAlpha(source.codeAt(nameStart))) { return -1; }
      if (namedCharacterReference(source.slice(nameStart, i + 1)) == null) { -1 } else { i + 1 }
    }

    let isAsciiHexDigit(c: Int): Boolean {
      isAsciiDigit(c) || (c >= char'a' && c <= char'f') || (c >= char'A' && c <= char'F')
    }

What a reference that `characterReferenceEnd` found stands for, given
it as written, from `&` to `;`. Anything else is given back as is.

    export let decodeCharacterReference(written: String): String {
      let source = new SourceText(written);
      let n = source.length;
      if (n < 3 || source.codeAt(0) != char'&' || source.codeAt(n - 1) != char';') { return written; }
      if (source.codeAt(1) != char'#') {
        return namedCharacterReference(source.slice(1, n)) ?? written;
      }
      let hex = source.codeAt(2) == char'x' || source.codeAt(2) == char'X';
      var value = 0;
      for (var i = if (hex) { 3 } else { 2 }; i < n - 1; i += 1) {
        let c = source.codeAt(i);
        let digit = if (isAsciiDigit(c)) {
          c - char'0'
        } else if (c >= char'a' && c <= char'f') {
          c - char'a' + 10
        } else if (c >= char'A' && c <= char'F') {
          c - char'A' + 10
        } else {
          return written;
        };
        value = value * (if (hex) { 16 } else { 10 }) + digit;
      }
      codePointOrReplacement(value)
    }

## Tests

    test("named character references, with and without semicolons") {
      assert(namedCharacterReference("amp;") == "&") { "amp;" };
      assert(namedCharacterReference("amp") == "&") { "legacy amp" };
      assert(namedCharacterReference("ouml;") == "ö") { "ouml;" };
      assert(namedCharacterReference("ouml") == "ö") { "legacy ouml" };
      assert(namedCharacterReference("Abreve;") == "Ă") { "Abreve;" };
      assert(namedCharacterReference("Abreve") == null) { "no legacy Abreve" };
      assert(namedCharacterReference("ngE;") == "≧̸") { "two code points" };
      assert(namedCharacterReference("fjlig;") == "fj") { "letters" };
      assert(namedCharacterReference("NewLine;") == "\n") { "a line feed" };
      assert(namedCharacterReference("ClockwiseContourIntegral;") == "∲") { "long" };
      assert(namedCharacterReference("zwnj;") == "\u{200C}") { "the last" };
      assert(namedCharacterReference("AElig;") == "Æ") { "the first" };
      assert(namedCharacterReference("bogus;") == null && namedCharacterReference("") == null) { "none" };
      assert(namedCharacterReference("AMP;") == "&" && namedCharacterReference("amp;;") == null) { "case" };
    }

    test("finding and decoding character references") {
      let cases = [
        ["&amp;", "&"], ["&#35;", "#"], ["&#1234;", "Ӓ"], ["&#X22;", "\""], ["&#xcab;", "ಫ"],
        ["&#0;", "\u{FFFD}"], ["&#xD800;", "\u{FFFD}"], ["&#x110000;", "\u{FFFD}"],
        ["&#9999999;", "\u{FFFD}"], ["&HilbertSpace;", "ℋ"],
      ];
      for (var i = 0; i < cases.length; i += 1) {
        let written = cases[i][0];
        let source = new SourceText("${written}x");
        let end = characterReferenceEnd(source, 0, source.length);
        assert(end == source.length - 1) { "${written}: ${end.toString()}" };
        assert(decodeCharacterReference(written) == cases[i][1]) { decodeCharacterReference(written) };
      }
      let bad = ["&nbsp", "&x;", "&#;", "&#x;", "&#87654321;", "&#abcdef0;", "&ThisIsNotDefined;", "&hi?;", "&1x;", "&"];
      for (var i = 0; i < bad.length; i += 1) {
        let source = new SourceText(bad[i]);
        assert(characterReferenceEnd(source, 0, source.length) == -1) { bad[i] };
      }
    }

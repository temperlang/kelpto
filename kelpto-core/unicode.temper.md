# Unicode classes

What CommonMark's inline rules need from Unicode, as its spec (0.31.2)
defines them, read from the tables in `unicode-tables.temper.md`
(generated by `scripts/unicode_tables.py`, which says how they're
encoded). ASCII is answered without the tables.

**Unicode whitespace:** general category Zs, plus tab, line feed, form
feed, and carriage return.

    export let isUnicodeWhitespace(c: Int): Boolean {
      if (c < 0x80) {
        return c == char' ' || c == char'\t' || c == char'\n' || c == 0xC || c == char'\r';
      }
      for (var i = 0; i < whitespaceRanges.length; i += 2) {
        if (c < whitespaceRanges[i]) { return false; }
        if (c <= whitespaceRanges[i + 1]) { return true; }
      }
      false
    }

**Unicode punctuation:** general categories P (punctuation) and S
(symbols), so `$`, `+`, `<`, `€`, and `©` count as well as `!` and `«`.

    export let isUnicodePunctuation(c: Int): Boolean {
      if (c < 0x80) {
        return (c >= 0x21 && c <= 0x2F) || (c >= 0x3A && c <= 0x40) ||
          (c >= 0x5B && c <= 0x60) || (c >= 0x7B && c <= 0x7E);
      }
      let k = chunkOf(punctuationStarts, c);
      if (k < 0) { return false; }
      let text = punctuationRuns[k];
      var at = punctuationStarts[k];
      var i = String.begin;
      var value = 0;
      var start = -1;
      while (text.hasIndex(i)) {
        let d = text[i];
        i = text.next(i);
        value = (value << 5) | (d & 0x1F);
        if (d < 0x60) {
          if (start < 0) {
            start = at + value;
            if (c < start) { return false; }
          } else {
            at = start + value;
            if (c < at) { return true; }
            start = -1;
          }
          value = 0;
        }
      }
      false
    }

The last chunk starting at or before `c`, or -1.

    let chunkOf(starts: List<Int>, c: Int): Int {
      if (c < starts[0]) { return -1; }
      var low = 0;
      var high = starts.length - 1;
      while (low < high) {
        let middle = (low + high + 1) / 2;
        if (starts[middle] <= c) { low = middle; } else { high = middle - 1; }
      }
      low
    }

**Case folding,** full (CaseFolding.txt's C and F entries), as
CommonMark matches link labels: `ẞ` and `ß` fold to `ss`, and `K` (the
Kelvin sign) to `k`. Text that isn't valid Unicode, such as a lone
surrogate, stays as it was.

    export let caseFold(text: String): String {
      let out = new StringBuilder();
      var runStart = String.begin;
      var i = String.begin;
      while (text.hasIndex(i)) {
        let c = text[i];
        let next = text.next(i);
        if (c >= char'A' && c <= char'Z') {
          out.append(text.slice(runStart, i));
          out.appendCodePoint(c + 32) orelse panic();
          runStart = next;
        } else if (c >= 0x80 && foldInto(c, null)) {
          out.append(text.slice(runStart, i));
          foldInto(c, out);
          runStart = next;
        }
        i = next;
      }
      out.append(text.slice(runStart, i));
      out.toString()
    }

Whether `c` folds to something else; if so, and `out` isn't null, the
fold is appended to it.

    let foldInto(c: Int, out: StringBuilder?): Boolean {
      let k = chunkOf(foldStarts, c);
      if (k < 0) { return false; }
      let text = foldRuns[k];
      var at = foldStarts[k];
      var i = String.begin;
      // The numbers of the entry being read, as they come.
      let numbers = new ListBuilder<Int>();
      var value = 0;
      while (text.hasIndex(i)) {
        let d = text[i];
        i = text.next(i);
        value = (value << 5) | (d & 0x1F);
        if (d < 0x60) {
          numbers.add(value);
          value = 0;
          let n = numbers.length;
          // Not `if (n == 1) { if (…) { … } } else if`: C# drops the
          // braces around the inner `if`, so the `else` goes with it.
          if (n == 1 && c < at + numbers[0]) {
            return false;
          } else if (n >= 3 && numbers[1] == 0 && n == 3 + numbers[2]) {
            let start = at + numbers[0];
            if (c == start) {
              if (out != null) {
                for (var j = 3; j < n; j += 1) {
                  out.appendCodePoint(start + unzigzag(numbers[j])) orelse panic();
                }
              }
              return true;
            }
            at = start + 1;
            numbers.clear();
          } else if (n == 4 && numbers[1] > 0) {
            let start = at + numbers[0];
            let stride = numbers[2];
            at = start + (numbers[1] - 1) * stride + 1;
            if (c < at) {
              if (stride == 2 && (c - start) % 2 != 0) { return false; }
              if (out != null) {
                out.appendCodePoint(c + unzigzag(numbers[3])) orelse panic();
              }
              return true;
            }
            numbers.clear();
          }
        }
      }
      false
    }

    let unzigzag(n: Int): Int {
      if (n % 2 == 0) { n / 2 } else { -(n / 2) - 1 }
    }

## Tests

    test("Unicode whitespace and punctuation") {
      assert(isUnicodeWhitespace(char' ') && isUnicodeWhitespace(0xC) && isUnicodeWhitespace(0xA0)) { "spaces" };
      assert(isUnicodeWhitespace(0x3000) && isUnicodeWhitespace(0x202F)) { "Zs" };
      assert(!isUnicodeWhitespace(0xB) && !isUnicodeWhitespace(0x200B) && !isUnicodeWhitespace(0x85)) {
        "VT, zero width space, and NEL aren't"
      };
      assert(isUnicodePunctuation(char'!') && isUnicodePunctuation(char'$') && isUnicodePunctuation(char'~')) {
        "ASCII"
      };
      assert(!isUnicodePunctuation(char'a') && !isUnicodePunctuation(char' ') && !isUnicodePunctuation(char'0')) {
        "not punctuation"
      };
      // « (Pi), € (Sc), © (So), ¬ (Sm), ‿ (Pc), and the first and last of a range.
      let yes = [0xAB, 0x20AC, 0xA9, 0xAC, 0x203F, 0xA1, 0xA7, 0x1F600, 0x1FBFA];
      for (var i = 0; i < yes.length; i += 1) {
        assert(isUnicodePunctuation(yes[i])) { yes[i].toString(16) };
      }
      // A letter, a mark, a space, unassigned, and past the last.
      let no = [0xAA, 0xE9, 0x301, 0xA0, 0x378, 0x1FBFB, 0x10FFFF];
      for (var i = 0; i < no.length; i += 1) {
        assert(!isUnicodePunctuation(no[i])) { no[i].toString(16) };
      }
    }

    test("case folding, full") {
      let cases = [
        ["ABC xyz", "abc xyz"], ["ẞ", "ss"], ["ß", "ss"], ["\u{212A}", "k"], ["ΣΑΣ", "σασ"],
        ["ﬃ", "ffi"], ["İ", "i\u{307}"], ["Ǆ", "ǆ"], ["Ā ā Ă", "ā ā ă"], ["𐐀", "𐐨"],
        ["already folded: é", "already folded: é"], ["", ""],
      ];
      for (var i = 0; i < cases.length; i += 1) {
        let folded = caseFold(cases[i][0]);
        assert(folded == cases[i][1]) { "${cases[i][0]}: ${folded}" };
      }
    }
